! Cisco Router R4 Configuration (FLSM 192.168.10.0/27)
enable

 description LAN-4 (Branch-South)
 no shutdown
exit

 description WAN-Link-R4-to-R3
 no shutdown
exit

 ip address 192.168.10.225 255.255.255.224
 description WAN-Link-R4-to-R1
 no shutdown
exit
